#ps1_sysnative
# YallaCloud deploy-time userdata - FROZEN. Same verbatim content on every Windows
# template 2016-2025. No secrets, nothing to edit. The virtual router serves this over
# plain HTTP and it stays readable by any local process, so it must never carry one.
# The first line above must survive: lose it and nothing runs while the VM looks fine.
$ErrorActionPreference = 'Continue'
$log = 'C:\Scripts\yc-userdata.log'
function L($m){ "$(Get-Date -f 'yyyy-MM-dd HH:mm:ss')  $m" | Out-File $log -Append -Encoding ascii }
L '=== yc userdata start ==='

# --- Fix 1: Administrator lockout (ALL templates) -------------------------
try { & net accounts /lockoutthreshold:0 | Out-Null; L 'lockout threshold -> 0' } catch { L "lockout FAILED: $_" }

# --- Fix 2: TLS 1.2 - SERVER 2016 ONLY ------------------------------------
try {
  $cap = (Get-CimInstance Win32_OperatingSystem).Caption
  if ($cap -match '2016') {
    foreach($p in @('HKLM:\SOFTWARE\Microsoft\.NETFramework\v4.0.30319',
                    'HKLM:\SOFTWARE\WOW6432Node\Microsoft\.NETFramework\v4.0.30319')){
      if(-not (Test-Path $p)){ New-Item $p -Force | Out-Null }
      New-ItemProperty $p -Name SystemDefaultTlsVersions -Value 1 -PropertyType DWord -Force | Out-Null
      New-ItemProperty $p -Name SchUseStrongCrypto      -Value 1 -PropertyType DWord -Force | Out-Null
    }
    $s = 'HKLM:\SYSTEM\CurrentControlSet\Control\SecurityProviders\SCHANNEL\Protocols\TLS 1.2\Client'
    if(-not (Test-Path $s)){ New-Item $s -Force | Out-Null }
    New-ItemProperty $s -Name Enabled           -Value 1 -PropertyType DWord -Force | Out-Null
    New-ItemProperty $s -Name DisabledByDefault -Value 0 -PropertyType DWord -Force | Out-Null
    L "TLS 1.2 enabled (Server 2016: $cap)"
  } else { L "TLS 1.2 skipped - not Server 2016 ($cap)" }
} catch { L "TLS FAILED: $_" }

# --- Fix 3: console logon focuses Administrator ---------------------------
try {
  $u = 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Authentication\LogonUI'
  if(-not (Test-Path $u)){ New-Item $u -Force | Out-Null }
  New-ItemProperty $u -Name LastLoggedOnUser    -Value '.\Administrator' -PropertyType String -Force | Out-Null
  New-ItemProperty $u -Name LastLoggedOnSAMUser -Value '.\Administrator' -PropertyType String -Force | Out-Null
  L 'console logon user -> .\Administrator'
} catch { L "logonui FAILED: $_" }

# --- Fix 5: network context - DETECT, never assume ------------------------
try {
  $cfg = Get-CimInstance Win32_NetworkAdapterConfiguration | Where-Object { $_.IPEnabled -and $_.DHCPServer } | Select-Object -First 1
  $vr  = $cfg.DHCPServer
  $gw  = ($cfg.DefaultIPGateway | Select-Object -First 1)
  $kind = if ($vr -and $gw -and $vr -eq $gw) { 'isolated' } else { 'shared' }
  L ("net: ip={0} gw={1} vr={2} dns={3} kind={4}" -f ($cfg.IPAddress -join ','), $gw, $vr, ($cfg.DNSServerSearchOrder -join ','), $kind)
  if (-not $gw)  { L 'net: WARNING no default gateway from DHCP' }
} catch { L "net context FAILED: $_" }

# --- Fix 4: C:\Scripts sync from the frozen public URLs -------------------
$U = 'https://raw.githubusercontent.com/yallacloud/yc-scripts/main/YallaCloud-CScripts-latest.zip'
$S = 'https://raw.githubusercontent.com/yallacloud/yc-scripts/main/YallaCloud-CScripts-latest.sha256'
try {
  [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
  $net = $false
  try { $net = (Test-NetConnection -ComputerName 'raw.githubusercontent.com' -Port 443 -WarningAction SilentlyContinue).TcpTestSucceeded } catch {}
  if (-not $net) {
    try { $c=New-Object Net.Sockets.TcpClient; $ia=$c.BeginConnect('raw.githubusercontent.com',443,$null,$null)
          $net=$ia.AsyncWaitHandle.WaitOne(8000,$false); $c.Close() } catch { $net=$false }
  }
  if (-not $net) {
    L 'kit sync SKIPPED - no HTTPS egress to raw.githubusercontent.com:443'
  } elseif (-not (Test-Path 'C:\Scripts\Update-YcScripts.ps1')) {
    L 'kit sync SKIPPED - C:\Scripts\Update-YcScripts.ps1 missing (must be baked in before sealing)'
  } else {
    L 'kit sync: internet OK, running Update-YcScripts'
    & 'C:\Scripts\Update-YcScripts.ps1' -Url $U -ShaUrl $S *>&1 | Out-File $log -Append -Encoding ascii
    L ("Update-YcScripts exit={0}  (0 ok|2 args|3 zip missing|4 SHA mismatch|5 not admin|6 extract|7 gate)" -f $LASTEXITCODE)
  }
} catch { L "kit sync FAILED: $_" }

# --- Fix 6: fix-deploy - OUTSIDE the internet gate ------------------------
# Fix-Deploy is a LOCAL script. Only the sync above needs egress. Running it
# inside the gate meant an isolated VM never got the Administrator UNLOCK or
# the isolated-network gateway fix - the two fixes not done inline above -
# because the whole branch was skipped for having no internet.
try {
  if (Test-Path 'C:\Scripts\Fix-Deploy.ps1') {
    & 'C:\Scripts\Fix-Deploy.ps1' *>&1 | Out-File $log -Append -Encoding ascii
    L ("fix-deploy exit={0}" -f $LASTEXITCODE)
  } else { L 'fix-deploy SKIPPED - C:\Scripts\Fix-Deploy.ps1 not present' }
} catch { L "fix-deploy FAILED: $_" }

L '=== yc userdata end ==='
